{{- /* Overrides themes/loveit/layouts/partials/home/profile.html. Only the avatar differs. The theme renders it through plugin/img.html, which emits a lazysizes placeholder: The real image is swapped in by JavaScript. Two problems with that here: 1. It is the hero image, above the fold. Lazy-loading it saves nothing and makes the most visible image on a site that sells websites depend on a third-party script running successfully. 2. static/images/avatar.png is 1.2 MB, sent in full to every visitor including phones, to be displayed at 8rem. So: a plain , no JS involved, resized by Hugo at build time from assets/images/avatar.png, with width/height set to avoid layout shift. Everything below the avatar is a verbatim copy of the theme partial. */ -}} {{- $profile := .Site.Params.home.profile -}}
{{- $avatar := $profile.avatarURL -}} {{- with $profile.gravatarEmail -}} {{- $avatar = md5 . | printf "https://www.gravatar.com/avatar/%v?s=240&d=mp" -}} {{- end -}} {{- if $avatar -}} {{- /* Build the avatar once, then reuse it inside or outside the link. */ -}} {{- $avatarImg := "" -}} {{- $alt := .Site.Params.author.name | default .Site.Title -}} {{- with resources.Get "images/avatar.png" -}} {{- $small := .Resize "256x webp q82" -}} {{- $large := .Resize "512x webp q82" -}} {{- $avatarImg = printf `` $small.RelPermalink $small.RelPermalink $large.RelPermalink $small.Width $small.Height $alt -}} {{- else -}} {{- /* Gravatar or a path outside assets/: plain img, still no lazyload. */ -}} {{- $avatarImg = printf `` $avatar $alt -}} {{- end -}}
{{- $menus := $.Site.Menus.main | default slice -}} {{- with index $menus 0 -}} {{- $url := .URL | relLangURL -}} {{- with .Page -}} {{- $url = .RelPermalink -}} {{- end -}} {{- $avatarImg | safeHTML -}} {{- else -}} {{- $avatarImg | safeHTML -}} {{- end -}}
{{- end -}} {{- with $profile.title -}}

{{- . | safeHTML -}}
